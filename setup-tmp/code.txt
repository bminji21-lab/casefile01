var SHEET_ID = '1maFiYUz-j6IYks-yNovCbMRSaszkviVV7VhGHil1l_0'; // 기록용 구글 시트
function book() { return SpreadsheetApp.openById(SHEET_ID); }

var CONFIG = {
  SITE: 'https://lawimpact.github.io',
  DEADLINE: new Date('2026-11-14T00:00:00+09:00'), // 11/13(금) 23:59까지 인증
  MISSIONS: {
    1: { prefix: 'LB01', name: 'LAWBTI' },
    2: { prefix: 'CF01', name: 'CASE FILE 01' },
    3: { prefix: 'CF02', name: 'CASE FILE 02' }
  },
  TZ: 'Asia/Seoul'
};

var SHEET_PEOPLE = '참가자';
var SHEET_LOG = '인증기록';
var PEOPLE_HEAD = ['토큰', '이름', '학교', '휴대폰', 'M1 LAWBTI', 'M2 CASE FILE 01', 'M3 CASE FILE 02', '완료 수', '완주', '처음 인증', '마지막 인증', '메모'];
var LOG_HEAD = ['시각', '미션', '이름', '학교', '휴대폰', '인증코드', '유형', '코드 확인', '토큰', '경로', '결과'];

/* ---------- 처음 한 번 실행 ---------- */
function setup() {
  var ss = book();
  var people = ss.getSheetByName(SHEET_PEOPLE) || ss.insertSheet(SHEET_PEOPLE, 0);
  var log = ss.getSheetByName(SHEET_LOG) || ss.insertSheet(SHEET_LOG, 1);
  [[people, PEOPLE_HEAD], [log, LOG_HEAD]].forEach(function (p) {
    var sh = p[0], head = p[1];
    sh.getRange(1, 1, 1, head.length).setValues([head])
      .setFontWeight('bold').setBackground('#b8e3f6');
    sh.setFrozenRows(1);
  });
  people.getRange('D:D').setNumberFormat('@');
  log.getRange('E:E').setNumberFormat('@');
  people.setColumnWidth(1, 110);
  // 완주자만 한눈에: 완주 칸에 색
  var rule = SpreadsheetApp.newConditionalFormatRule()
    .whenTextEqualTo('완주').setBackground('#aee67e')
    .setRanges([people.getRange('I2:I')]).build();
  people.setConditionalFormatRules([rule]);
  var guide = ss.getSheetByName('안내');
  if (guide) { guide.getRange('A1').setValue('참가자 탭: 한 사람당 한 줄. 인증기록 탭: 인증이 들어올 때마다 한 줄씩 쌓여요. 완주 칸이 연두색이면 굿즈 대상.'); }
}

/* ---------- 인증코드 확인 (게임과 같은 계산식) ---------- */
var ALPHA = '23456789ABCDEFGHJKLMNPQRSTUVWXYZ';
function fnv(s) {
  var h = 0x811c9dc5;
  for (var i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 0x01000193); }
  return h >>> 0;
}
function norm(s) { return String(s || '').replace(/\s+/g, '').toLowerCase(); }
function h4(name, school) {
  var h = fnv(norm(name) + '|' + norm(school)), out = '';
  for (var i = 0; i < 4; i++) { out += ALPHA[h & 31]; h >>>= 5; }
  return out;
}
function checkDigit(h, type, fast) { return String(fnv(h + type + (fast ? '#S' : '#N') + '#LAFA') % 10); }
function verifyCode(m, name, school, code) {
  var mission = CONFIG.MISSIONS[m];
  if (!mission) { return false; }
  var re = new RegExp('^' + mission.prefix + '-([' + ALPHA + ']{4})-([A-Z])(\\d)$');
  var hit = re.exec(String(code || '').trim().toUpperCase());
  if (!hit) { return false; }
  if (hit[1] !== h4(name, school)) { return false; }
  return hit[3] === checkDigit(hit[1], hit[2], true) || hit[3] === checkDigit(hit[1], hit[2], false);
}

/* ---------- 공통 ---------- */
function json(obj, callback) {
  var text = JSON.stringify(obj);
  if (callback && /^[A-Za-z_$][\w$]*$/.test(callback)) {
    return ContentService.createTextOutput(callback + '(' + text + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(text).setMimeType(ContentService.MimeType.JSON);
}
function now() { return Utilities.formatDate(new Date(), CONFIG.TZ, 'yyyy-MM-dd HH:mm:ss'); }
function short(d) { return d ? String(d).slice(5, 10).replace('-', '.') : ''; }
function cleanPhone(p) { return String(p || '').replace(/\D/g, ''); }
function newToken() {
  var chars = 'abcdefghjkmnpqrstuvwxyz23456789', t = '';
  for (var i = 0; i < 10; i++) { t += chars[Math.floor(Math.random() * chars.length)]; }
  return t;
}
function sheets() {
  var ss = book();
  return { people: ss.getSheetByName(SHEET_PEOPLE), log: ss.getSheetByName(SHEET_LOG) };
}
function findRow(sheet, col, value) {
  var last = sheet.getLastRow();
  if (last < 2) { return -1; }
  var vals = sheet.getRange(2, col, last - 1, 1).getDisplayValues();
  for (var i = 0; i < vals.length; i++) { if (String(vals[i][0]) === String(value)) { return i + 2; } }
  return -1;
}
function cardOf(row, people) {
  var v = people.getRange(row, 1, 1, PEOPLE_HEAD.length).getDisplayValues()[0];
  var done = [], when = {};
  [1, 2, 3].forEach(function (m) { if (v[3 + m]) { done.push(m); when[m] = short(v[3 + m]); } });
  return { ok: true, token: v[0], name: v[1], school: v[2], done: done, when: when };
}

/* ---------- 웹 인증 ---------- */
function submit(d, via) {
  var m = Number(d.m), name = String(d.name || '').trim(), school = String(d.school || '').trim();
  var phone = cleanPhone(d.phone), code = String(d.code || '').trim().toUpperCase(), type = String(d.type || '').trim();
  var s = sheets();
  var logRow = [now(), m, name, school, phone, code, type, '', '', via, ''];

  if (new Date() >= CONFIG.DEADLINE) { logRow[10] = '마감 후'; s.log.appendRow(logRow); return { ok: false, error: 'closed', message: '사전미션 인증이 마감됐어요.' }; }
  if (!CONFIG.MISSIONS[m] || !name || !school) { return { ok: false, error: 'bad', message: '미션 정보가 비어 있어요. 미션 마지막 화면에서 다시 눌러 주세요.' }; }
  if (!/^01\d{8,9}$/.test(phone)) { return { ok: false, error: 'phone', message: '휴대폰 번호를 확인해 주세요. (예: 010-1234-5678)' }; }

  var okCode = verifyCode(m, name, school, code);
  logRow[7] = okCode ? '정상' : '불일치';
  if (!okCode) { logRow[10] = '거절'; s.log.appendRow(logRow); return { ok: false, error: 'code', message: '인증코드가 이름·학교와 맞지 않아요. 미션에서 입력한 이름·학교 그대로인지 확인해 주세요.' }; }

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var row = findRow(s.people, 4, phone), memo = '';
    if (row < 0) {
      s.people.appendRow([newToken(), name, school, phone, '', '', '', '', '', now(), now(), '']);
      row = s.people.getLastRow();
      s.people.getRange(row, 8, 1, 2).setFormulas([['=COUNTA(E' + row + ':G' + row + ')', '=IF(H' + row + '=3,"완주","")']]);
    } else {
      var cur = s.people.getRange(row, 2, 1, 2).getDisplayValues()[0];
      if (norm(cur[0]) !== norm(name) || norm(cur[1]) !== norm(school)) {
        memo = 'M' + m + '에서 다른 이름/학교로 인증: ' + name + ' / ' + school;
        var old = s.people.getRange(row, 12).getDisplayValue();
        s.people.getRange(row, 12).setValue(old ? old + '\n' + memo : memo);
      }
    }
    var cell = s.people.getRange(row, 4 + m);
    var already = !!cell.getDisplayValue();
    if (!already) { cell.setValue(now()); }
    s.people.getRange(row, 11).setValue(now());
    var card = cardOf(row, s.people);
    logRow[8] = card.token; logRow[10] = already ? '이미 받음' : '도장';
    s.log.appendRow(logRow);
    card.fresh = already ? 0 : m;
    return card;
  } finally {
    lock.releaseLock();
  }
}

/* ---------- 카카오 챗봇 스킬 (챗봇 연결 후) ---------- */
function kakao(body) {
  var text = String((body.userRequest && body.userRequest.utterance) || '');
  var mName = /이름:\s*([^/\n]+)/.exec(text), mSchool = /학교:\s*([^/\n]+)/.exec(text);
  var mCode = /인증코드:\s*([A-Z0-9-]+)/i.exec(text);
  var reply;
  if (mName && mSchool && mCode) {
    var name = mName[1].trim(), school = mSchool[1].trim(), code = mCode[1].trim().toUpperCase();
    var m = 0;
    Object.keys(CONFIG.MISSIONS).forEach(function (k) { if (code.indexOf(CONFIG.MISSIONS[k].prefix + '-') === 0) { m = Number(k); } });
    var url = CONFIG.SITE + '/cert/?m=' + m + '&n=' + encodeURIComponent(name) + '&s=' + encodeURIComponent(school) + '&c=' + encodeURIComponent(code);
    reply = {
      version: '2.0',
      template: { outputs: [{ basicCard: {
        title: (CONFIG.MISSIONS[m] ? CONFIG.MISSIONS[m].name : '사전미션') + ' 인증 확인!',
        description: name + ' (' + school + ')\n아래 버튼을 눌러 휴대폰 번호를 넣으면 도장이 찍혀요.',
        buttons: [{ action: 'webLink', label: '도장 받으러 가기', webLinkUrl: url }]
      } }] }
    };
  } else {
    reply = { version: '2.0', template: { outputs: [{ simpleText: { text: '사전미션 마지막 화면의 인증 문구를 그대로 붙여넣어 보내 주세요.' } }] } };
  }
  return ContentService.createTextOutput(JSON.stringify(reply)).setMimeType(ContentService.MimeType.JSON);
}

/* ---------- 입구 ---------- */
function doPost(e) {
  var body = {};
  try { body = JSON.parse((e.postData && e.postData.contents) || '{}'); } catch (err) { body = {}; }
  if (body.userRequest) { return kakao(body); }
  try {
    if (body.action === 'submit') { return json(submit(body, 'web')); }
    return json({ ok: false, error: 'unknown' });
  } catch (err) {
    return json({ ok: false, error: 'server', message: '잠시 후 다시 시도해 주세요.' });
  }
}

function doGet(e) {
  var p = (e && e.parameter) || {}, cb = p.callback;
  try {
    if (p.k) {
      var s = sheets(), row = findRow(s.people, 1, p.k);
      return json(row < 0 ? { ok: false, error: 'notfound' } : cardOf(row, s.people), cb);
    }
    return json({ ok: true, service: 'law-impact-stamp' }, cb);
  } catch (err) {
    return json({ ok: false, error: 'server' }, cb);
  }
}
